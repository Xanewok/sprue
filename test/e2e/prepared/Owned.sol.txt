pragma solidity ^0.8.27;

contract Owned {
    address public immutable OWNER;
    uint256 public fee;
    uint256 public cap;

    constructor(address owner) {
        OWNER = owner;
    }

    function setFee(uint256 v) external {
        require(msg.sender == OWNER, "not the owner");
        fee = v;
    }

    function setCap(uint256 v) external {
        require(msg.sender == OWNER, "not the owner");
        cap = v;
    }
}
